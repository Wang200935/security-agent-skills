def solve_lll(M):
    """Generic LLL wrapper — many implementations available."""
    from sympy import Matrix
    M_sympy = Matrix(M.tolist())
    reduced = M_sympy.LLL()
    return [[int(x) for x in row] for row in reduced.tolist()]
